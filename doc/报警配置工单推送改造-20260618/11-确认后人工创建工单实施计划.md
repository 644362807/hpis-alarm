# 报警确认后人工创建工单 Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

> **执行前修订提示（2026-06-23）：** 本计划是在 `13-报警确认工单闭环详细设计方案.md` 之前生成的，不能直接按旧稿执行。进入代码实现前，必须先按 `13` 号文档补充或修订以下内容：报警确认权限继续走报警管理权限；`alarm_status` 不承载确认、处理、已转工单状态；创建工单页面默认带出模板负责人并允许确认人临时改派；`alarm_workorder.assignee_id/name` 保存实际负责人；工单关闭后必须写 `alarm_handle` 工单同步动作；已创建工单后不再走报警侧直接处理完成。

**Goal:** 在不改 `alarm` 主表职责的前提下，实现“报警确认后人工创建工单、工单负责人通知、报警详情通过 `alarm_handle` 展示工单绑定”的最小可行闭环。

**Architecture:** `alarm` 只保存报警事实；`alarm_handle` 保存确认、处理、创建工单等动作流水；`alarm_workorder` 保存工单事实。创建工单是确认后的人工动作，服务端按当前报警配置的 `workorder_config_id` 校验，使用 `alarm_workorder.alarm_id` 唯一约束保证幂等。

**Tech Stack:** Java 8, Spring Boot, MyBatis/MyBatis-Plus, MySQL 8, Redis, RabbitMQ, ShardingSphere time-capacity sharding, JUnit.

---

## Scope

本文只记录后续代码实现计划，不代表本轮已经修改 Java、XML、SQL 或配置文件。

本计划只覆盖 `hpis-alarm` 最小闭环，以及使用现有推送通道发送工单创建通知所需的报警侧 payload。`hpis-push` 的行业级路由增强、全局推送路由、复杂工单流转引擎不进入本轮。

本轮不做：

- 不给 `alarm` 新增 `workorder_id`。
- 不给 `alarm` 新增 `alarm_configure_id`。
- 不新增 `config_version`。
- 不自动创建工单。
- 不改变 RabbitMQ ack/nack 语义。
- 不改变 stop 消警主链路。

CREATE PROCEDURE add_column_if_missing(IN table_name_param varchar(128), IN column_name_param varchar(128), IN alter_sql_param varchar(1000))
BEGIN
  IF NOT EXISTS (
    SELECT 1
    FROM information_schema.columns
    WHERE table_schema = DATABASE()
      AND table_name = table_name_param
      AND column_name = column_name_param
  ) THEN
    SET @sql_text = CONCAT('ALTER TABLE `', table_name_param, '` ', alter_sql_param);
    PREPARE stmt FROM @sql_text;
    EXECUTE stmt;
    DEALLOCATE PREPARE stmt;
  END IF;
END$$
DELIMITER ;

package com.hpis.alarm.service.support;

import com.hpis.alarm.domain.Alarm;

public class AlarmConfigureResolver {

    public AlarmConfigureResolver(AlarmConfigureMapper alarmConfigureMapper) {
        this.alarmConfigureMapper = alarmConfigureMapper;
    }

    public AlarmConfigure resolveForAlarm(Alarm alarm) {
        if (alarm == null || alarm.getTenantId() == null
                || alarm.getSceneType() == null || alarm.getDeviceSn() == null || alarm.getAlarmType() == null) {
            return null;
        }
        AlarmConfigure query = new AlarmConfigure();
        query.setTenantId(alarm.getTenantId());
        query.setSceneType(alarm.getSceneType());
        query.setDeviceSn(alarm.getDeviceSn());
        query.setAlarmType(alarm.getAlarmType());
        return alarmConfigureMapper.selectEnabledDeviceRule(query);
    }
}

import org.springframework.dao.DuplicateKeyException;
import org.springframework.stereotype.Service;
import org.springframework.transaction.annotation.Transactional;

import java.util.List;

@Service
public class AlarmWorkorderServiceImpl implements IAlarmWorkorderService {
    private static final Snowflake SNOWFLAKE = new Snowflake(7, 7);

    private final AlarmMapper alarmMapper;
    private final AlarmWorkorderMapper alarmWorkorderMapper;
    private final AlarmWorkorderConfigMapper alarmWorkorderConfigMapper;
    private final AlarmHandleMapper alarmHandleMapper;
    private final AlarmConfigureResolver alarmConfigureResolver;

    public AlarmWorkorderServiceImpl(AlarmMapper alarmMapper,
                                     AlarmWorkorderMapper alarmWorkorderMapper,
                                     AlarmWorkorderConfigMapper alarmWorkorderConfigMapper,
                                     AlarmHandleMapper alarmHandleMapper,
                                     AlarmConfigureResolver alarmConfigureResolver) {
        this.alarmMapper = alarmMapper;
        this.alarmWorkorderMapper = alarmWorkorderMapper;
        this.alarmWorkorderConfigMapper = alarmWorkorderConfigMapper;
        this.alarmHandleMapper = alarmHandleMapper;
        this.alarmConfigureResolver = alarmConfigureResolver;
    }

    @Override
    @Transactional(rollbackFor = Exception.class)
    public AlarmWorkorderCreateResult createAfterConfirm(AlarmWorkorderCreateRequest request) {
        if (request == null || request.getAlarmId() == null) {
            throw new IllegalArgumentException("alarmId不能为空");
        }
        AlarmWorkorder existed = alarmWorkorderMapper.selectByAlarmId(request.getAlarmId());
        if (existed != null) {
            return AlarmWorkorderCreateResult.existed(existed.getWorkorderId());
        }
        Alarm alarm = alarmMapper.selectAlarmById(request.getAlarmId());
        if (alarm == null) {
            throw new IllegalArgumentException("报警不存在，alarmId=" + request.getAlarmId());
        }
        AlarmConfigure alarmConfigure = alarmConfigureResolver.resolveForAlarm(alarm);
        if (alarmConfigure == null || alarmConfigure.getWorkorderConfigId() == null) {
            throw new IllegalStateException("当前报警配置未关联工单配置，不能创建工单");
        }
        AlarmWorkorderConfig config = alarmWorkorderConfigMapper.selectEnabledById(alarmConfigure.getWorkorderConfigId());
        if (config == null) {
            throw new IllegalStateException("工单配置未启用或不存在，workorderConfigId=" + alarmConfigure.getWorkorderConfigId());
        }
        Long workorderId = SNOWFLAKE.nextId();
        AlarmWorkorder workorder = buildWorkorder(workorderId, alarm, config);
        try {
            alarmWorkorderMapper.insertAlarmWorkorder(workorder);
        } catch (DuplicateKeyException duplicateKeyException) {
            AlarmWorkorder concurrent = alarmWorkorderMapper.selectByAlarmId(request.getAlarmId());
            if (concurrent != null) {
                return AlarmWorkorderCreateResult.existed(concurrent.getWorkorderId());
            }
            throw duplicateKeyException;
        }
        insertCreateWorkorderAction(alarm, workorderId);
        return AlarmWorkorderCreateResult.created(workorderId);
    }

    @Override
    public AlarmWorkorder selectByAlarmId(Long alarmId) {
        return alarmWorkorderMapper.selectByAlarmId(alarmId);
    }

    @Override
    public List<AlarmWorkorder> selectAlarmWorkorderList(AlarmWorkorder alarmWorkorder) {
        return alarmWorkorderMapper.selectAlarmWorkorderList(alarmWorkorder);
    }

    private AlarmWorkorder buildWorkorder(Long workorderId, Alarm alarm, AlarmWorkorderConfig config) {
        AlarmWorkorder workorder = new AlarmWorkorder();
        workorder.setWorkorderId(workorderId);
        workorder.setWorkorderNo("AW" + workorderId);
        workorder.setAlarmId(alarm.getAlarmId());
        workorder.setWorkorderConfigId(config.getWorkorderConfigId());
        workorder.setWorkorderType(config.getWorkorderType());
        workorder.setWorkorderPriority(config.getWorkorderPriority());
        workorder.setTitle(config.getTitleTemplate() == null ? "报警工单" : config.getTitleTemplate());
        workorder.setContent(config.getContentTemplate() == null ? "请处理报警" : config.getContentTemplate());
        workorder.setDeviceSn(alarm.getDeviceSn());
        workorder.setAlarmType(alarm.getAlarmType());
        workorder.setAlarmRank(alarm.getAlarmRank());
        workorder.setAlarmTime(alarm.getAlarmBegintime());
        workorder.setAssigneeId(config.getAssigneeId());
        workorder.setAssigneeName(config.getAssigneeName());
        workorder.setStatus(AlarmWorkorderStatusEnums.WAIT_ACCEPT.getKey());
        workorder.setTenantId(alarm.getTenantId());
        workorder.setDelFlag("0");
        workorder.setCreateTime(DateUtils.getNowDate());
        return workorder;
    }

    private void insertCreateWorkorderAction(Alarm alarm, Long workorderId) {
        AlarmHandle action = new AlarmHandle();
        action.setAlarmId(alarm.getAlarmId());
        action.setAlarmBegintime(alarm.getAlarmBegintime());
        action.setHandleType(AlarmHandleTypeEnums.CREATE_WORKORDER.getKey());
        action.setWorkorderId(workorderId);
        action.setHandleStatus(alarm.getAlarmStatus());
        action.setDelFlag("0");
        action.setCreateTime(DateUtils.getNowDate());
        action.setHandleTime(DateUtils.getNowDate());
        alarmHandleMapper.insertAlarmHandleAction(action);
    }
}
