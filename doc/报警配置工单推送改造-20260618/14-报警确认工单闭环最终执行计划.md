# 报警确认工单闭环 Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 在不修改 `alarm` 主表职责的前提下，实现报警确认后可直接处理或创建工单、工单可改派负责人、工单关闭同步报警详情时间线的最小闭环。

**Architecture:** `alarm` 只保存报警事实；`alarm_configure` 保存推送策略编码和可选工单模板；`alarm_handle` 保存确认、直接处理、创建工单、工单同步动作；`alarm_workorder` 保存工单事实。创建工单由确认后用户点击触发，负责人来自模板默认值但允许临时改派，一条报警最多一个工单。

**Tech Stack:** Java 8, Spring Boot, MyBatis/MyBatis-Plus, MySQL, Redis, RabbitMQ, ShardingSphere time-capacity sharding, JUnit.

```sql
ALTER TABLE alarm_configure
  ADD COLUMN push_message_type varchar(30) NULL COMMENT '推送策略编码，对应推送配置message_type',
  ADD COLUMN workorder_config_id bigint NULL COMMENT '关联报警工单模板ID，为空表示确认后不能按模板创建工单';

ALTER TABLE alarm_handle
  ADD COLUMN handle_type char(2) NULL COMMENT '动作类型：0确认 1直接处理 2关闭 3误报 4创建工单 5工单同步',
  ADD COLUMN workorder_id bigint NULL COMMENT '关联工单ID',
  ADD COLUMN status_before char(2) NULL COMMENT '动作前状态',
  ADD COLUMN status_after char(2) NULL COMMENT '动作后状态';

CREATE TABLE IF NOT EXISTS alarm_workorder_config (
  workorder_config_id bigint NOT NULL AUTO_INCREMENT COMMENT '工单模板ID',
  config_name varchar(100) NOT NULL COMMENT '工单模板名称',
  enabled char(1) DEFAULT '1' COMMENT '是否启用：0否 1是',
  workorder_type varchar(50) NULL COMMENT '工单类型',
  workorder_priority char(2) NULL COMMENT '工单优先级',
  assignee_type varchar(30) NULL COMMENT '派单类型：USER/ROLE/DEVICE_OWNER/AREA_OWNER',
  assignee_id bigint NULL COMMENT '模板默认负责人ID',
  assignee_name varchar(100) NULL COMMENT '模板默认负责人名称快照',
  title_template varchar(200) NULL COMMENT '工单标题模板',
  content_template varchar(1000) NULL COMMENT '工单内容模板',
  tenant_id bigint NULL COMMENT '租户ID',
  del_flag char(2) DEFAULT '0' COMMENT '逻辑删除：0存在 2删除',
  create_by varchar(64) NULL COMMENT '创建人',
  create_time datetime DEFAULT CURRENT_TIMESTAMP COMMENT '创建时间',
  update_by varchar(64) NULL COMMENT '更新人',
  update_time datetime NULL DEFAULT NULL ON UPDATE CURRENT_TIMESTAMP COMMENT '更新时间',
  remark varchar(500) NULL COMMENT '备注',
  PRIMARY KEY (workorder_config_id),
  KEY idx_workorder_config_tenant (tenant_id, enabled, del_flag)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COMMENT='报警工单模板表';

CREATE TABLE IF NOT EXISTS alarm_workorder (
  workorder_id bigint NOT NULL COMMENT '工单ID',
  workorder_no varchar(50) NOT NULL COMMENT '工单编号',
  alarm_id bigint NOT NULL COMMENT '报警ID',
  workorder_config_id bigint NULL COMMENT '来源工单模板ID',
  workorder_type varchar(50) NULL COMMENT '工单类型',
  workorder_priority char(2) NULL COMMENT '工单优先级',
  title varchar(200) NULL COMMENT '工单标题',
  content varchar(1000) NULL COMMENT '工单内容',
  device_sn varchar(100) NULL COMMENT '设备SN快照',
  alarm_type char(2) NULL COMMENT '报警类型快照',
  alarm_rank char(2) NULL COMMENT '报警等级快照',
  alarm_time datetime NULL COMMENT '报警时间快照',
  assignee_id bigint NULL COMMENT '当前实际负责人ID',
  assignee_name varchar(100) NULL COMMENT '当前实际负责人名称',
  status char(2) DEFAULT '0' COMMENT '状态：0待接单 1处理中 2已处理 3已关闭 4退回 5取消',
  handle_result varchar(500) NULL COMMENT '工单处理结果',
  tenant_id bigint NULL COMMENT '租户ID',
  del_flag char(2) DEFAULT '0' COMMENT '逻辑删除：0存在 2删除',
  create_by varchar(64) NULL COMMENT '工单创建人',
  create_time datetime DEFAULT CURRENT_TIMESTAMP COMMENT '创建时间',
  update_by varchar(64) NULL COMMENT '更新人',
  update_time datetime NULL DEFAULT NULL ON UPDATE CURRENT_TIMESTAMP COMMENT '更新时间',
  remark varchar(500) NULL COMMENT '备注',
  PRIMARY KEY (workorder_id),
  UNIQUE KEY uk_alarm_workorder_no (workorder_no),
  UNIQUE KEY uk_alarm_workorder_alarm (alarm_id),
  KEY idx_alarm_workorder_config (workorder_config_id),
  KEY idx_alarm_workorder_status (tenant_id, status, create_time),
  KEY idx_alarm_workorder_assignee (assignee_id, status)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COMMENT='报警工单表';
```
